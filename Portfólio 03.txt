programa
{
    inclua biblioteca Util --> u
    inclua biblioteca Calendario --> c

    funcao inicio()
    {
        inteiro numero

        // Sorteia um número entre 1 e 100
        numero = u.sorteia(1, 100)

        escreva("Sorteando um número...\n")

        // Aguarda 5 segundos
        u.aguarde(5000)

        // Exibe o número sorteado
        escreva("Número sorteado: ", numero, "\n")

        // Exibe a data da geração
        escreva("Dia: ", c.dia_mes_atual(), "\n")
    }
}
